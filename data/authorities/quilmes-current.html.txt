<script>
(function() {
  var header = document.getElementById('site-header');
  var toggler = document.getElementById('site-navbar-toggler');
  var menu = document.getElementById('site-navbar-menu');
  if (!header || !toggler || !menu) return;

  function isMobile() { return window.innerWidth <= 768; }

  function setMobileMenuPosition() {
    var preheader = document.querySelector('.site-preheader');
    var navbar = document.querySelector('.site-navbar');
    if (!navbar) return;
    var top = (window.scrollY > 0)
      ? navbar.offsetHeight
      : (preheader ? preheader.offsetHeight + navbar.offsetHeight : navbar.offsetHeight);
    document.documentElement.style.setProperty('--mobile-header-height', top + 'px');
  }
  function openMenu() {
    setMobileMenuPosition();
    document.body.classList.add('site-nav-open');
    document.body.style.overflow = 'hidden';
    header.classList.add('has-menu-open');
    toggler.setAttribute('aria-label', 'Cerrar menú');
    toggler.setAttribute('aria-expanded', 'true');
  }
  function closeMenu() {
    document.body.classList.remove('site-nav-open');
    document.body.style.overflow = '';
    header.classList.remove('has-menu-open');
    toggler.setAttribute('aria-label', 'Abrir menú');
    toggler.setAttribute('aria-expanded', 'false');
    menu.querySelectorAll('li.is-expanded').forEach(function(li) { li.classList.remove('is-expanded'); });
  }
  function toggleMenu() {
    if (header.classList.contains('has-menu-open')) closeMenu();
    else openMenu();
  }

  toggler.addEventListener('click', function() {
    if (isMobile()) toggleMenu();
  });

  menu.addEventListener('click', function(e) {
    if (!isMobile()) return;
    var a = e.target.closest('a');
    if (!a || !menu.contains(a)) return;
    var li = a.closest('li');
    var sub = li ? li.querySelector(':scope > .dropdown') : null;
    var isParentLink = li && li.querySelector(':scope > a') === a;
    if (isParentLink && sub) {
      e.preventDefault();
      li.classList.toggle('is-expanded');
    } else {
      closeMenu();
    }
  });

  window.addEventListener('resize', function() {
    if (!isMobile()) closeMenu();
    else if (header.classList.contains('has-menu-open')) setMobileMenuPosition();
  });

  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape' && header.classList.contains('has-menu-open')) closeMenu();
  });

  function normalizeNavPath(pathname) {
    var path = (pathname || '/').replace(/\/+$/, '');
    return (path === '' ? '/' : path).toLowerCase();
  }

  function normalizeHost(host) {
    return (host || '').replace(/^www\./i, '').toLowerCase();
  }

  function hrefToPath(href) {
    if (!href || href.charAt(0) === '#') return null;
    try {
      return new URL(href, window.location.origin);
    } catch (err) {
      return null;
    }
  }

  function isNavLinkActive(anchor) {
    var href = anchor.getAttribute('href');
    var linkUrl = hrefToPath(href);
    if (!linkUrl) return false;
    if (normalizeHost(linkUrl.hostname) !== normalizeHost(window.location.hostname)) return false;
    var linkPath = normalizeNavPath(linkUrl.pathname);
    var currentPath = normalizeNavPath(window.location.pathname);
    if (linkPath === currentPath) return true;
    if (href.charAt(href.length - 1) === '/' && linkPath !== '/') {
      return currentPath === linkPath || currentPath.indexOf(linkPath + '/') === 0;
    }
    return false;
  }

  menu.querySelectorAll('a[href]').forEach(function(anchor) {
    if (isNavLinkActive(anchor)) {
      anchor.classList.add('is-active');
      var parentItem = anchor.closest('.site-navbar-menu > li');
      if (parentItem && parentItem.querySelector(':scope > .dropdown') && anchor.closest('.dropdown')) {
        parentItem.classList.add('is-expanded');
      }
    }
  });
})();
